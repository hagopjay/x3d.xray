Three examples of the exploratory stage

Each one runs offline on synthetic data (seeded, no clock, no network). What is printed below is what the code printed when this page was built, and CI runs the same three files as tests, so the page cannot drift from the code. Every number here comes from the exact-integer EDA kernel in x3dray/eda.py (and, for example 3, its JavaScript twin in x3dray/web/tools.js).

1. Signal and noise

What does a Confirmed grade mean, and how often does pure noise earn one?

#!/usr/bin/env python3
"""Example 1: what a Confirmed look is, and what it is not.

Two histories are run through the same twelve declared looks. In one, Claude's commits are planted to be five times smaller and to fall in a
narrower band of hours. In the other nothing is planted at all. The planted effects come out Confirmed, as they should. Then the noise: of the
six pure-noise histories below, one still earns a Confirmed (the window of seeds was chosen to include it; the unselected rate is the calibration line below). That is not a bug. The grade is a robustness grade (the pattern survived the declared
checks, at q <= 1/20), not a probability that the pattern is real, and a few noise histories in a hundred will earn one, which is why the answer
prints how many looks were taken and the notes say "a pattern that survived every declared check in this history; not causal, not predictive".

Run: python3 examples/eda_1_signal_and_noise.py            (about 30 s)
     python3 examples/eda_1_signal_and_noise.py --calibrate (recomputes the 150-history calibration, about 6 minutes)
"""
from __future__ import annotations

import sys

from _common import eda, synthetic_dataset, table

PLANTED = dict(seed=7, repos=4, per_repo=150, plant=("ai_hours", "ai_smaller"))
NOISE_SEEDS = tuple(range(1028, 1034))               # seed 1031 is the one that earns a Confirmed
NOISE = dict(repos=3, per_repo=120)
CALIBRATION_SEEDS = range(1000, 1150)
RECORDED_CALIBRATION = "4 of 150 histories (seeds 1000-1149) earned at least one Confirmed; recorded in the eda PR, not recomputed unless --calibrate"


def looks_of(**kw) -> dict:
    section, problems = eda.run(synthetic_dataset(**kw))
    assert problems == [], problems
    return section["looks"]


def summary(looks: dict) -> dict:
    graded = [f for f in looks["findings"] if f["grade"]]
    return {"by_grade": dict(looks["ledger"]["by_grade"]), "graded": [(f["id"], f["grade"], f["grade_reason"], f["effect"]["permille"], f["q_text"]) for f in graded]}


def run(calibrate: bool = False) -> dict:
    planted = summary(looks_of(**PLANTED))
    noise = {seed: summary(looks_of(seed=seed, **NOISE)) for seed in NOISE_SEEDS}
    out = {"planted": planted, "noise": noise, "noise_with_confirmed": [s for s, r in noise.items() if r["by_grade"]["C"]]}
    if calibrate:
        out["calibration"] = sum(1 for s in CALIBRATION_SEEDS if looks_of(seed=s, **NOISE)["ledger"]["by_grade"]["C"])
    return out


def report(r: dict) -> str:
    lines = ["PLANTED: Claude's commits 5x smaller and in a narrower band of hours (synthetic, seed 7, 4 repos x 150 commits)"]
    lines.append(table([[i, g, why, f"{pm}/1000", q] for i, g, why, pm, q in r["planted"]["graded"]], ["look", "grade", "why", "effect", "q"]))
    lines.append(f"  ledger: {r['planted']['by_grade']}\n")
    lines.append(f"NOISE: nothing planted (synthetic, 3 repos x 120 commits, seeds {NOISE_SEEDS[0]}-{NOISE_SEEDS[-1]})")
    lines.append(table([[s, v["by_grade"]["C"], v["by_grade"]["L"], v["by_grade"]["S"], v["by_grade"]["none"]] for s, v in r["noise"].items()],
                       ["seed", "Confirmed", "Likely", "Suspected", "no grade"]))
    for seed in r["noise_with_confirmed"]:
        lines.append(f"\n  seed {seed} earned a Confirmed from nothing:")
        lines.append("  " + table([[i, g, why, f"{pm}/1000", q] for i, g, why, pm, q in r["noise"][seed]["graded"]], ["look", "grade", "why", "effect", "q"]).replace("\n", "\n  "))
    lines.append("\nCALIBRATION: " + (f"{r['calibration']} of 150 histories (seeds 1000-1149) earned at least one Confirmed (recomputed now)" if "calibration" in r else RECORDED_CALIBRATION))
    lines.append("Read it as: a Confirmed means the pattern survived the declared checks, with q <= 1/20 over all twelve looks. A few noise histories will still earn one.")
    return "\n".join(lines)


def main(argv=None) -> int:
    print(report(run("--calibrate" in (argv if argv is not None else sys.argv[1:]))))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

2. Simpson's paradox

What happens when a difference between two kinds of author is really a difference between repos?

The groups differ a lot and q is tiny, yet nothing is Confirmed, because no repo holds both groups. The grade refuses to turn a between-repo difference into a claim about authors.

CONFOUNDED: the author class follows the repo (synthetic, seed 9, 4 repos x 150 commits)
look            effect     q       grade     why                             repo strata
--------------  ---------  ------  --------  ------------------------------  -------------------------
contrast:lines  -589/1000  0.0060  L         no_stratum_with_both_groups     no repo holds both groups
contrast:hour4  76/1000    0.9310  no grade  effect_below_floor              no repo holds both groups
contrast:repo   1000/1000  0.0060  L         leave_one_repo_out_unavailable  no repo holds both groups

ledger: {'C': 0, 'L': 2, 'S': 0, 'none': 10}
Read it as: the difference between the groups is large and its q is small, yet nothing is Confirmed. Strata and leave-one-repo-out are part of the grade,
so a difference that lives between repos is never promoted to a finding about authors. contrast:repo is 1000/1000 because the repo and the class are the same variable here.
Show the source (eda_2_simpsons_paradox.py)
#!/usr/bin/env python3
"""Example 2: when the repo explains the difference, the grade says so.

In this synthetic history the author class follows the repo: one repo is all human commits and the others all Claude commits. Across the whole
family Claude's commits look very different from the human ones (they are in different repos, and repos differ). Nothing inside any one repo
differs, because no repo has both kinds of author. A tool that only compared the two groups would call that a finding. This one grades it Likely
at best, and names the reason: no repo held both groups, so the difference cannot be separated from the repo.

Run: python3 examples/eda_2_simpsons_paradox.py            (about 10 s)
"""
from __future__ import annotations

from _common import eda, synthetic_dataset, table

DATA = dict(seed=9, repos=4, per_repo=150, plant="confound")
FOCUS = ("contrast:lines", "contrast:hour4", "contrast:repo")


def run() -> dict:
    section, problems = eda.run(synthetic_dataset(**DATA))
    assert problems == [], problems
    looks = section["looks"]
    by_id = {f["id"]: f for f in looks["findings"]}
    return {"ledger": looks["ledger"]["by_grade"], "pair": looks["pair"], "focus": {i: by_id[i] for i in FOCUS}, "audit_ok": problems == []}


def report(r: dict) -> str:
    rows = []
    for i, f in r["focus"].items():
        s = f["checks"]["strata"]
        rows.append([i, f"{f['effect']['permille']}/1000", f["q_text"], f["grade"] or "no grade", f["grade_reason"],
                     "no repo holds both groups" if s and s["n"] == 0 or s is None else f"{s['same']} of {s['n']} repos agree"])
    return "\n".join([
        "CONFOUNDED: the author class follows the repo (synthetic, seed 9, 4 repos x 150 commits)",
        table(rows, ["look", "effect", "q", "grade", "why", "repo strata"]),
        f"\nledger: {r['ledger']}",
        "Read it as: the difference between the groups is large and its q is small, yet nothing is Confirmed. Strata and leave-one-repo-out are part of the grade,",
        "so a difference that lives between repos is never promoted to a finding about authors. contrast:repo is 1000/1000 because the repo and the class are the same variable here."])


def main() -> int:
    print(report(run()))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

3. The price of looking

Why is the same answer worth less when it is the fortieth look than when it is the thirteenth?

The same p, nine different q. Every look you take is added to the family the correction counts over. This one runs the page's own JavaScript, so it is also the check that the browser and the Python agree.

THE PRICE OF LOOKING (synthetic, seed 7: 'you' vs 'ai' in one repo, UTC four-hour block, all commits)
other looks first  q counted over m =  p (add-one)  q       against 1/20
-----------------  ------------------  -----------  ------  ----------------
0                  13                  0.0020       0.0130  clears q <= 1/20
1                  14                  0.0020       0.0140  clears q <= 1/20
2                  15                  0.0020       0.0150  clears q <= 1/20
4                  17                  0.0020       0.0170  clears q <= 1/20
8                  21                  0.0020       0.0210  clears q <= 1/20
16                 29                  0.0020       0.0290  clears q <= 1/20
24                 37                  0.0020       0.0370  clears q <= 1/20
31                 44                  0.0020       0.0440  clears q <= 1/20
38                 51                  0.0020       0.0510  does not clear

one answer: Alice, in r1 vs Bot-AI, in r1, UTC four-hour block: distribution distance 0.324 (largest gap in 16-19); 1 of 999 label shuffles were at least as extreme (p 0.0020, q 0.0130 over m = 13 looks); ungraded.
Read it as: p never moves, since the data and the permutation count are the same. q moves, because m does: every look you take is added to the family.
Show the source (eda_3_the_price_of_looking.py)
#!/usr/bin/env python3
"""Example 3: the same question, asked later, is worth less.

One question about one synthetic history ("do human and Claude commits in one repo fall in different four-hour blocks?") is asked after k other
looks have been taken. The data do not change and neither does the permutation count behind it: the add-one p is 0.0020 every time. What changes
is q, because q is corrected over the 12 declared looks plus every look taken in the session, and the answer prints that denominator. By the
time about 38 other looks have been taken, the same answer no longer clears q <= 1/20. This is the tool counting the garden of forking paths for
you. The count is also what the page's notebook shows, and what the investigation chain records.

It runs the page's own tools.js in Node (the same code Ask Claude and the MCP server use), so it needs `node`.

Run: python3 examples/eda_3_the_price_of_looking.py            (about 5 s)
"""
from __future__ import annotations

from _common import eda, minimal_bundle, node_available, run_node, synthetic_dataset, table

DATA = dict(seed=7, repos=4, per_repo=150, plant=("ai_hours",))
PRIOR_LOOKS = (0, 1, 2, 4, 8, 16, 24, 31, 38)
SCRIPT = r'''
const ks = JSON.parse(process.argv[2]);
const mk = () => { const seen = new Set(); let asked = 0;
  return { note(l) { const k = JSON.stringify(l); if (seen.has(k)) return; seen.add(k); if (l.kind !== 'declared') asked++; }, get asked() { return asked; }, settle: async () => null, unrecorded: [] }; };
const run = (t, a) => t.find(x => x.name === 'eda').execute(a);
const decoys = [];
for (let h = 0; h < 24; h++) decoys.push({ topic: 'slice', hour: h, measure: 'lines' });
for (let w = 0; w < 7; w++) decoys.push({ topic: 'slice', weekday: String(w), measure: 'lines' });
for (const r of B.repos) decoys.push({ topic: 'slice', repo: r.name, measure: 'lines' });
for (const r of B.repos) for (const m of ['files', 'gap_s']) decoys.push({ topic: 'slice', repo: r.name, measure: m });
const rows = [];
for (const k of ks) {
  const t = T.makeTools(B, null, { looks: mk() });
  for (let i = 0; i < k; i++) run(t, decoys[i]);
  const r = run(t, { topic: 'compare', a: { who: 'you', repo: 'r1' }, b: { who: 'ai', repo: 'r1' }, measure: 'hour4', fork: 'all' });
  rows.push({ prior: k, m: r.family.m, p: r.null_model.p_text, q: r.q_text, asked: r.looks_asked_this_session, text: r.text });
}
console.log(JSON.stringify({ rows, decoys: decoys.length }));
'''


def run() -> dict:
    if not node_available():
        raise RuntimeError("this example runs the page's tools.js and needs node")
    section, problems = eda.run(synthetic_dataset(**DATA))
    assert problems == [], problems
    out = run_node(SCRIPT, minimal_bundle(synthetic_dataset(**DATA), section), __import__("json").dumps(list(PRIOR_LOOKS)))
    out["limit"] = "q <= 1/20"
    return out


def report(r: dict) -> str:
    rows = [[x["prior"], x["m"], x["p"], x["q"], "clears q <= 1/20" if float(x["q"]) <= 0.05 else "does not clear"] for x in r["rows"]]
    return "\n".join(["THE PRICE OF LOOKING (synthetic, seed 7: 'you' vs 'ai' in one repo, UTC four-hour block, all commits)",
                      table(rows, ["other looks first", "q counted over m =", "p (add-one)", "q", "against 1/20"]),
                      f"\none answer: {r['rows'][0]['text']}",
                      "Read it as: p never moves, since the data and the permutation count are the same. q moves, because m does: every look you take is added to the family."])


def main() -> int:
    print(report(run()))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
